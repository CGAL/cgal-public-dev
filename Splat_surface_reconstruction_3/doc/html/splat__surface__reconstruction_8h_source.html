215  std::vector<Vector_3> compute_block_normals() const {
216 
217  std::vector<Vector_3> block_normals;
218  block_normals.reserve(cells_.size());
219 
220  for (std::size_t ix = 0; ix < nx_; ++ix) {
221  for (std::size_t iy = 0; iy < ny_; ++iy) {
222  for (std::size_t iz = 0; iz < nz_; ++iz) {
223  const Cell& c = cells_[flat_index(ix, iy, iz)];
224 
225  // If no points contributed to this cell, we leave the normal as NULL_VECTOR.
226  if (c.normal_count == 0) {
227  block_normals.push_back(CGAL::NULL_VECTOR);
228  continue;
229  }
230 
231  // Average the normals of points in this cell to get the block normal.
232  const FT inv = FT(1) / FT(c.normal_count);
233  Vector_3 n = c.normal_sum * inv;
234 
235  // Normalize the block normal to have unit length.
236  const double len2 = CGAL::to_double(n.squared_length());
237  block_normals.push_back(n * (1.0 / std::sqrt(len2)));
238  }
239  }
240  }
241 
242  return block_normals;
243  }

258  std::vector<FT> estimate_individual_splat_sizes(FT global_spacing) const {
259  std::vector<FT> splat_sizes(points_.size(), global_spacing);
260 
261  for (Index i = 0; i < points_.size(); ++i) {
262  int cx, cy, cz;
263  if (!to_grid_coords(points_[i], cx, cy, cz)) {
264  std::cerr << "Warning: point " << i << " is out of grid bounds, skipping splat size estimation.\n";
265  continue;
266  }
267 
268  // Collect all points in the same box as p_i.
269  std::vector<Index> local_ids;
270  const Cell* c = cell(cx, cy, cz);
271  if (c == nullptr || c->point_ids.empty()) {
272  std::cerr << "Warning: no points found in the same cell for point " << i << ", skipping splat size estimation.\n";
273  continue;
274  }
275  local_ids = c->point_ids;
276 
277  // Need at least a few neighbors to define a local triangulation.
278  if (local_ids.size() < 3) {
279  std::cerr << "Warning: not enough neighbors for point " << i << ", skipping splat size estimation.\n";
280  splat_sizes[i] = global_spacing;
281  continue;
282  }
283 
284  // Build a local tangent frame at p_i from its normal.
285  std::vector<Vector_3> local_frame = compute_local_tangent_frame(points_[i], normals_[i]);
286  Vector_3 u = local_frame[0];
287  Vector_3 v = local_frame[1];
288 
289  // Project local neighbors onto the tangent plane and compute their 2D coordinates
290  // Also input in the Delaunay Triangulation Kernel
291  CGAL::Delaunay_triangulation_2<Kernel> DT;
292  DT.insert(Point_2(0, 0)); // insert the center point itself at the origin of the local frame
293  for (Index neighbor_id : local_ids) {
294  if (neighbor_id == i) continue; // skip the center point
295  Vector_3 vec = points_[neighbor_id] - points_[i];
296  double x = CGAL::to_double(vec * u);
297  double y = CGAL::to_double(vec * v);
298  Point_2 point_2d(x, y);
299 
300  DT.insert(point_2d);
301  }
302 
303  //Find the circumcenter of each triangle in Delaunay triangulation
304  std::vector<FT> circumcenter_distances;
305  for (auto face = DT.finite_faces_begin(); face != DT.finite_faces_end(); ++face) {
306  Point_2 c = CGAL::circumcenter(face->vertex(0)->point(), face->vertex(1)->point(), face->vertex(2)->point());
307  double dist = std::sqrt(CGAL::to_double(c.x()*c.x() + c.y()*c.y())); // distance from circumcenter to the origin (which is the projection of p_i)
308  circumcenter_distances.push_back(FT(dist));
309  }
310 
311  splat_sizes[i] = *std::max_element(circumcenter_distances.begin(), circumcenter_distances.end());
312  }
313 
314  return splat_sizes;
315  }

325  std::vector<Vector_3> compute_local_tangent_frame(const Point_3& p, const Vector_3& n) const {
326  // Find a vector that is not parallel to n to construct the tangent frame.
327  Vector_3 temp;
328  if (std::abs(CGAL::to_double(n.x())) < 0.9)
329  temp = Vector_3(1,0,0);
330  else
331  temp = Vector_3(0,1,0);
332 
333  Vector_3 u = CGAL::cross_product(n, temp);
334  Vector_3 v = CGAL::cross_product(n, u);
335 
336  // Normalize u and v to have unit length.
337  const double len_u = std::sqrt(CGAL::to_double(u.squared_length()));
338  const double len_v = std::sqrt(CGAL::to_double(v.squared_length()));
339  if (len_u > 0) u = u / len_u;
340  if (len_v > 0) v = v / len_v;
341 
342  return {u, v};
343  }

415  y < miny || y > maxy ||
416  z < minz || z > maxz) {
417  return false;
418  }

421  iy = static_cast<int>(std::floor((y - miny) / h));
422  iz = static_cast<int>(std::floor((z - minz) / h));
423 
424  ix = std::min(ix, static_cast<int>(nx_) - 1);
425  iy = std::min(iy, static_cast<int>(ny_) - 1);
426  iz = std::min(iz, static_cast<int>(nz_) - 1);

664  Vector_3 compute_cell_normal(int ix, int iy, int iz) const
665  {
666  const Cell& c = cells_[flat_index(ix, iy, iz)];
667  if (c.normal_count == 0) {
668  return CGAL::NULL_VECTOR;

670

672  Vector_3 n = c.normal_sum * inv;

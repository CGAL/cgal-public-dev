105  void build(const std::vector<Point_3>& points,
106  const std::vector<Vector_3>* normals = nullptr)
107  {
108  clear();
109  points_ = points;
110 
111  if (normals != nullptr) {
112  normals_ = *normals;
113  if (normals_.size() != points_.size()) {
114  normals_.resize(points_.size(), CGAL::NULL_VECTOR);
115  }
116  } else {
117  normals_.assign(points_.size(), CGAL::NULL_VECTOR);
118  }
119 
120  for (Index i = 0; i < points_.size(); ++i) {
121  const Vector_3* n = nullptr;
122  if (i < normals_.size() && normals_[i] != CGAL::NULL_VECTOR) {
123  n = &normals_[i];
124  }
125  insert(i, points_[i], n);
126  }

129  void insert(Index point_id, const Point_3& p, const Vector_3* n = nullptr)
130  {
131  int ix, iy, iz;
132  if (!to_grid_coords(p, ix, iy, iz)) {
133  return; // Outside the fixed domain.
134  }
135 
136  cells_[flat_index(ix, iy, iz)].add_point(point_id, n);
137 
138  if (point_id >= points_.size()) {
139  points_.resize(point_id + 1);
140  normals_.resize(point_id + 1, CGAL::NULL_VECTOR);
141  }
142  points_[point_id] = p;
143  if (n != nullptr) {
144  normals_[point_id] = *n;
145  }
146  }

184  // Return integer grid coords for a point. False if out of bounds.
185  bool to_grid_coords(const Point_3& p, int& ix, int& iy, int& iz) const
186  {
187  const double h = CGAL::to_double(box_size_);
188  if (h <= 0.0) {
189  return false;
190  }
191 
192  const double x = CGAL::to_double(p.x());
193  const double y = CGAL::to_double(p.y());
194  const double z = CGAL::to_double(p.z());
195 
196  const double minv = CGAL::to_double(min_coord_);
197  const double maxv = CGAL::to_double(max_coord_);
198 
199  if (x < minv || x > maxv || y < minv || y > maxv || z < minv || z > maxv) {
200  return false;
201  }
202 
203  ix = static_cast<int>(std::floor((x - minv) / h));
204  iy = static_cast<int>(std::floor((y - minv) / h));
205  iz = static_cast<int>(std::floor((z - minv) / h));
206 
207  // Clamp points exactly on the upper boundary.
208  ix = std::min(ix, static_cast<int>(nx_) - 1);
209  iy = std::min(iy, static_cast<int>(ny_) - 1);
210  iz = std::min(iz, static_cast<int>(nz_) - 1);
211 
212  return valid_coords(ix, iy, iz);
213  }

215  std::vector<Index> nearby_point_ids(const Point_3& p, FT radius) const
216  {
217  std::vector<Index> result;
218 
219  int cx, cy, cz;
220  if (!to_grid_coords(p, cx, cy, cz)) {
221  return result;
222  }
223 
224  const int r = static_cast<int>(
225  std::ceil(CGAL::to_double(radius) / CGAL::to_double(box_size_))
226  );
227 
228  for (int dx = -r; dx <= r; ++dx) {
229  for (int dy = -r; dy <= r; ++dy) {
230  for (int dz = -r; dz <= r; ++dz) {
231  const int ix = cx + dx;
232  const int iy = cy + dy;
233  const int iz = cz + dz;
234  if (!valid_coords(ix, iy, iz)) {
235  continue;
236  }
237  const Cell& c = cells_[flat_index(ix, iy, iz)];
238  result.insert(result.end(), c.point_ids.begin(), c.point_ids.end());
239  }
240  }
241  }
242 
243  return result;
244  }

246  Vector_3 compute_box_normal(int ix, int iy, int iz) const
247  {
248  const Cell* c = cell(ix, iy, iz);
249  if (c == nullptr || c->normal_count == 0) {
250  return CGAL::NULL_VECTOR;
251  }
252 
253  const FT inv = FT(1) / FT(c->normal_count);
254  Vector_3 n = c->normal_sum * inv;
255 
256  const double len2 = CGAL::to_double(n.squared_length());
257  if (len2 <= 0.0) {
258  return CGAL::NULL_VECTOR;
259  }
260 
261  return n * (1.0 / std::sqrt(len2));
262  }

264  Vector_3 compute_box_normal(const Point_3& p) const

268  return CGAL::NULL_VECTOR;

288 private:
289  void rebuild_dimensions()
290  {
291  const double h = CGAL::to_double(box_size_);
292  const double extent = CGAL::to_double(max_coord_ - min_coord_);
293  const std::size_t n = static_cast<std::size_t>(std::ceil(extent / h));
294 
295  nx_ = ny_ = nz_ = std::max<std::size_t>(1, n);
296  cells_.clear();
297  cells_.resize(nx_ * ny_ * nz_);
298  }

300  bool valid_coords(int ix, int iy, int iz) const
301  {
302  return ix >= 0 && iy >= 0 && iz >= 0 &&
303  static_cast<std::size_t>(ix) < nx_ &&
304  static_cast<std::size_t>(iy) < ny_ &&
305  static_cast<std::size_t>(iz) < nz_;
306  }

308  std::size_t flat_index(int ix, int iy, int iz) const
309  {
310  return (static_cast<std::size_t>(ix) * ny_ + static_cast<std::size_t>(iy)) * nz_
311  + static_cast<std::size_t>(iz);
